function renderApp(data) { const appDiv = document.getElementById('app'); let html = ''; // 1. Render Manager of the Month Carousel with Team Names (Filtering out malformed or header rows) if (data.monthly && data.monthly.length > 0) { const activePeriods = data.monthly.filter(period => { if (!period.scores || period.scores.length === 0) return false; if (period.title.toLowerCase().includes('period period') || period.title.toLowerCase().includes('monthly side pot')) return false; const isMalformed = period.scores.every(s => String(s.score) === s.name); return !isMalformed; }); if (activePeriods.length > 0) { html += `
Manager of the Month
`; html += `
`; activePeriods.forEach((_, index) => { const isActive = index === activePeriods.length - 1 ? 'active' : ''; html += `
`; }); html += `
`; html += `
`; activePeriods.forEach(period => { let cleanTitle = period.title.replace(/^Period\s+\d+\s*\(([^)]+)\)/i, '$1'); html += `

${cleanTitle}

    ${period.scores.map((s, idx, arr) => { let scoreDisplay = String(s.score !== undefined && s.score !== null ? s.score : ''); if (idx === arr.length - 1 && !scoreDisplay.includes('💩')) { scoreDisplay += ' 💩'; } const teamName = TEAM_NAMES[s.name] || ''; return `
  • ${s.name}
    ${teamName ? `
    ${teamName}
    ` : ''}
    ${scoreDisplay}
    • `; }).join('')}
🏆 Winner: ${period.topScorer} (${period.winnerPrize})
🥈 Runner-up: ${period.runnerName} (${period.runnerPrize})
`; }); html += `
`; } } // 2. Render Overall Standings Table if (data.overall && data.overall.rows) { const headers = data.overall.headers; const playerIdx = headers.findIndex(h => h.toLowerCase().includes('player') || h.toLowerCase().includes('name') || h.toLowerCase().includes('manager')); const pointsIdx = headers.findIndex(h => h.toLowerCase().includes('point') || h.toLowerCase().includes('total')); const medalIdx = headers.findIndex(h => h.toLowerCase().includes('medal')); const weeksWonIdx = headers.findIndex(h => h.toLowerCase().includes('won') || h.toLowerCase().includes('week')); const motmIdx = headers.findIndex(h => h.toLowerCase().includes('motm') || h.toLowerCase().includes('award')); const prizeIdx = headers.findIndex(h => h.toLowerCase().includes('prize')); html += `
Overall Standings
${data.overall.rows.map(row => { const playerName = row[playerIdx !== -1 ? playerIdx : 0]; const points = row[pointsIdx !== -1 ? pointsIdx : 1]; const medal = medalIdx !== -1 ? row[medalIdx] : ''; const weeksWon = row[weeksWonIdx !== -1 ? weeksWonIdx : 4]; const motm = motmIdx !== -1 ? row[motmIdx] : 5; const prize = row[prizeIdx !== -1 ? prizeIdx : 6]; return ` `; }).join('')}
Manager Points Weeks Won MOTM Prize
${playerName} ${points} ${medal} ${weeksWon} ${motm} £${prize}
`; } // Last Updated Timestamp if (data.updated) { html += `

Last synced: ${data.updated}

`; } appDiv.innerHTML = html; // Automatically scroll carousel to the most recent active period on load const carousel = document.getElementById('period-carousel'); if (carousel) { carousel.scrollLeft = carousel.scrollWidth; carousel.addEventListener('scroll', () => { const cardWidth = carousel.querySelector('.carousel-card').offsetWidth + 16; const currentIndex = Math.round(carousel.scrollLeft / cardWidth); const dots = document.querySelectorAll('.dot'); dots.forEach((dot, idx) => { if (idx === currentIndex) { dot.classList.add('active'); } else { dot.classList.remove('active'); } }); }); } }
